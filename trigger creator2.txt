CREATE OR REPLACE FUNCTION notify_func()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.name IS DISTINCT FROM NEW.name THEN
        PERFORM pg_notify('name_update', 'db data changed in column name');
        PERFORM pg_notify(
    'name_update',
    json_build_object(
        'message', 'db data changed in column name',
        'old_value', OLD.name,
        'new_value', NEW.name
    )::text
);
    END IF;

    IF OLD.number IS DISTINCT FROM NEW.number THEN
        PERFORM pg_notify(
    'number_update',
    json_build_object(
        'message', 'db data changed in column number',
        'old_value', OLD.number,
        'new_value', NEW.number
    )::text
);
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE TRIGGER web1_data_update
AFTER UPDATE ON table1
FOR EACH ROW
EXECUTE FUNCTION notify_func();

CREATE OR REPLACE FUNCTION notify_func()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.name IS DISTINCT FROM NEW.name OR
       OLD.number IS DISTINCT FROM NEW.number
 THEN PERFORM pg_notify(
    'name_update',
    'db data changed in column | table1 | name | id: ' || NEW.id || ' | old_name_value: ' || OLD.name || ' | new_name_value: ' || NEW.name || ' | old_number_value: ' || OLD.number || ' | new_number_value: ' || NEW.number 
);
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE TRIGGER web1_data_update
AFTER UPDATE ON table1
FOR EACH ROW
EXECUTE FUNCTION notify_func();
